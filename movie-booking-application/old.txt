# =========================
# 1. Single Environment Variables with -e (Port 4000)
# =========================

# Basic run with custom database settings on port 4000
docker run -p 4000:4000 \
  -e SPRING_PROFILES_ACTIVE=prod \
  -e SPRING_DATASOURCE_PASSWORD=mySecretPassword \
  -e SPRING_H2_CONSOLE_ENABLED=false \
  -v movie-data:/app/data \
  movie-booking-app

# Access your app at: http://localhost:4000

# =========================
# 2. Multiple Environment Variables
# =========================

# Override multiple H2 and Spring Boot settings
docker run -p 4000:4000 \
  -e SPRING_PROFILES_ACTIVE=production \
  -e SERVER_PORT=8080 \
  -e SPRING_DATASOURCE_URL="jdbc:h2:file:/app/data/proddb;AUTO_SERVER=TRUE" \
  -e SPRING_DATASOURCE_USERNAME=admin \
  -e SPRING_DATASOURCE_PASSWORD=securePassword123 \
  -e SPRING_H2_CONSOLE_ENABLED=false \
  -e SPRING_JPA_HIBERNATE_DDL_AUTO=validate \
  -e SPRING_JPA_SHOW_SQL=false \
  -e JAVA_OPTS="-Xms512m -Xmx1024m" \
  -v movie-prod-data:/app/data \
  movie-booking-app

# =========================
# 3. Using Environment File (.env)
# =========================

# Create .env file first:
# SPRING_PROFILES_ACTIVE=staging
# SPRING_DATASOURCE_PASSWORD=stagingPassword
# SPRING_H2_CONSOLE_ENABLED=true
# SPRING_JPA_SHOW_SQL=true

docker run -p 4000:4000 --env-file .env -v movie-staging-data:/app/data  movie-booking-app

# =========================
# 4. Development Mode with H2 Console Access (Port 4000)
# =========================

# Run with H2 console enabled - Access at http://localhost:4000/h2-console
docker run -p 4000:4000 \
  -e SPRING_PROFILES_ACTIVE=dev \
  -e SPRING_H2_CONSOLE_ENABLED=true \
  -e SPRING_H2_CONSOLE_PATH=/h2-console \
  -e SPRING_JPA_SHOW_SQL=true \
  -e LOGGING_LEVEL_ROOT=DEBUG \
  -v movie-dev-data:/app/data \
  --name movie-booking-dev \
  movie-booking-app

# H2 Console: http://localhost:4000/h2-console

# =========================
# 5. Production Mode (Secure)
# =========================

# Production run with security hardening
docker run -d -p 4000:4000 \
  -e SPRING_PROFILES_ACTIVE=production \
  -e SPRING_DATASOURCE_PASSWORD=VerySecurePassword123! \
  -e SPRING_H2_CONSOLE_ENABLED=false \
  -e SPRING_JPA_HIBERNATE_DDL_AUTO=validate \
  -e SPRING_JPA_SHOW_SQL=false \
  -e MANAGEMENT_ENDPOINTS_WEB_EXPOSURE_INCLUDE=health,info \
  -e JAVA_OPTS="-Xms256m -Xmx512m -XX:+UseG1GC" \
  -v movie-prod-data:/app/data \
  --restart unless-stopped \
  --name movie-booking-prod \
  movie-booking-app

# =========================
# 6. Custom Database Location

# Run with custom database file location
docker run -p 4000:4000 \
  -e SPRING_DATASOURCE_URL="jdbc:h2:file:/custom/path/moviedb;AUTO_SERVER=TRUE" \
  -v /host/custom/path:/custom/path \
  -v movie-custom-data:/app/data \
  movie-booking-app

# =========================
# 9. Memory and Performance Tuning
# =========================

# Run with JVM performance tuning
docker run -p 4000:4000 \
  -e JAVA_OPTS="-Xms256m -Xmx1024m -XX:+UseG1GC -XX:+UseStringDeduplication" \
  -e SPRING_PROFILES_ACTIVE=performance \
  -v movie-perf-data:/app/data \
  --memory=1.5g \
  --cpus=2 \
  movie-booking-app
